Mathematical Model
Sum[j] = Σᵢ d(Member[i], Venue[j])
Max[j] = maxᵢ d(Member[i], Venue[j])
RouteCost[j] = d(Venue[j], r1) + d(r1, r2)
TotalTravel[j] = Sum[j] + RouteCost[j]

ΔT = (max(T) − min(T)) / mean(T)
ΔM = (max(X) − min(X)) / mean(X)

W_total = ΔT / (ΔT + ΔM)
W_max = ΔM / (ΔT + ΔM)

GlobalScore[j] = W_total × TotalTravel[j] + W_max × Max[j]
Winner = arg min(GlobalScore)
